# 1. Make sure all your changes are committed on your branch
git add .
git commit -m "Your commit message"

# 2. Push your branch to remote (if not already pushed)
git push origin your-branch-name

# 3. Switch to main branch
git checkout main

# 4. Pull the latest changes from main (important for team projects)
git pull origin main

# 5. Merge your branch into main
git merge your-branch-name

# 6. Push the updated main to remote
git push origin main
